        pwd_bytes = password.encode("utf-8")
        salt = bcrypt.gensalt()
        hashed = bcrypt.hashpw(pwd_bytes, salt)

El tiempo de caducidad del token está configurado para 15 minutos y la firma para generación de tokens se obtiene de las variables de entorno con lo cual se refuerza la seguridad del software.

\begin{lstlisting}[language=Python]
    def create_access_token(data: dict, expires_delta: Optional[timedelta]):
        to_encode = data.copy()
        if expires_delta:
            expire = datetime.now() + expires_delta
        else:
            expire = datetime.now() + timedelta(minutes=15)

        to_encode.update({"exp": expire})
        encoded_jwt = jwt.encode(to_encode, JWT_SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

\begin{lstlisting}[language=Python]
    class SelectHandler(BaseHandler):
        def handle(self, expression: exp.Select) -> str:
            match_clause = self._build_match(expression)
            where_clause = self.build_where(expression)
            return_clause = self._build_return(expression)

            order_clause = self._build_order_by(expression)
            limit_clause = self._build_limit(expression)

            parts = [match_clause, where_clause, return_clause, order_clause, limit_clause]
            return " ".join([p for p in parts if p]).strip()
\end{lstlisting}

\subparagraph{Implementación de ejecución de consultas:}
Con ayuda de la biblioteca \texttt{neo4j}, se implementó el ejecutor de consultas Cypher. Una vez que se ha traducido una consulta SQL, dicho resultado se ejecuta directamente hacia la base de datos Neo4j. El siguiente fragmento de código indica el proceso para interactuar con una base de datos en Neo4j. Primero, se crea una conexión hacia la base de datos objetivo, se traduce la consulta y se procesa la respuesta, devolviendo el resultado de la ejecución para ser utilizado en el backend.

\begin{lstlisting}[language=Python]
    class Neo4jExecutor:
        def execute(self, query: str) -> Dict[str, Any]:
            nodes = {}
            links = []
            with self.driver.session() as session:
                result = session.run(query)
                for record in result:
                    for key, value in record.items():
                        if hasattr(value, "labels"):
                            node_id = str(value.element_id or value.id)
                            if node_id not in nodes:
                                nodes[node_id] = {"id": node_id, "labels": list(value.labels), "properties": dict(value._properties)}
                        elif hasattr(value, "start_node"):  # Relación
                            links.append({"source": str(value.start_node.element_id or value.start_node.id), "target": str(value.end_node.element_id or value.end_node.id), "type": value.type, "properties": dict(value._properties)})
            return {"nodes": list(nodes.values()), "links": links}
\end{lstlisting}

\paragraph{Implementación de relaciones}
Dado que se estableció una base sólida con respecto a la traducción de sentencias \texttt{SELECT} durante el sprint anterior, se continuó con la extensión de su gramática especialmente al manejo de relaciones que se dan mediante el uso de \texttt{JOINs}. Para ello, se modificó el método que construye la cláusula \texttt{MATCH} con la cual se busca la tabla principal junto a su alias para transformarla en una etiqueta de nodo, luego se busca todos los JOINS que incluyan la sentencia y en caso que sea \texttt{LEFT JOIN} se utilizará \texttt{OPTIONAL MATCH}, de lo contrario para \texttt{INNER JOIN} se empleará solo \texttt{MATCH}. Posteriormente, se representa la relación apuntando al nodo con una etiqueta distinta que será su equivalente a la tabla secundaria. El siguiente fragmento de código indica el proceso de traducción de sentencias \texttt{SELECT}  que incluyen relaciones.

\begin{lstlisting}[language=Python]
    def _build_match_and_joins(self, expression: exp.Select) -> str:
        from_node = expression.find(exp.From)
        if not from_node:
            raise TranslationError("SELECT requiere una cláusula FROM.")

        main_table_exp = None
        if from_node.expressions:
            main_table_exp = from_node.expressions[0]
        elif from_node.this:
            main_table_exp = from_node.this
        if not main_table_exp:
            raise TranslationError("No se pudo detectar la tabla en la cláusula FROM.")

        main_table = self.get_table_name(main_table_exp)
        alias_candidate = getattr(main_table_exp, "alias", None)
        main_alias = alias_candidate or "n"

        if main_alias == main_table:
            main_alias = "n"

        match_parts = [f"MATCH ({main_alias}:{main_table})"]

        for join in expression.find_all(exp.Join):
            kind = join.kind or "INNER"
            target_exp = join.this
            target_name = self.get_table_name(target_exp)

            target_alias_candidate = getattr(target_exp, "alias", None)
            target_alias = target_alias_candidate or (
                target_name[0] if target_name else "t"
            )
            rel_name = f"RELATED_TO_{target_name.upper()}"
            prefix = "OPTIONAL MATCH" if "LEFT" in kind or "RIGHT" in kind else "MATCH"
            pattern = (
                f"{prefix} ({main_alias})-[:{rel_name}]->({target_alias}:{target_name})"
            )
            match_parts.append(pattern)

        return " ".join(match_parts)
\end{lstlisting}

\paragraph{Traducción de INSERT, UPDATE y DELETE}
Una vez extendida la grámatica de \texttt{SELECT}, se implementaron las reglas de mapeo para el resto de operaciones de manipulación de datos. A continuación se indican los equivalentes en Cypher para cada tipo de operaciones:
\begin{itemize}
    \item La cláusula \texttt{INSERT INTO} se traduce a una sentencia Cypher bajo la siguiente estructura: \texttt{CREATE (n:Etiqueta {prop: 'valor'})}. Esta sentencia le ayudará a construir un nuevo nodo con la etiqueta y propiedades correspondientes.
    \item La sentencia \texttt{UPDATE} utiliza la cláusula \texttt{MATCH} que seleccionará el nodo a ser modificado y \texttt{SET} que actualizará la propiedad indicada.
    \item Por último, \texttt{DELETE} también utiliza \texttt{MATCH} para obtener el nodo y \texttt{DELETE} para eliminar los nodos. En caso de que el nodo disponga de asociaciones se utilizará \texttt{DETACH DELETE}.
\end{itemize}

\paragraph{Obtención y Exposición de Estadísticas}
\subparagraph{Backend:}
Se desarrolló un nuevo servicio de estadísticas encargado de calcular diversas métricas del sistema. Este servicio realiza consultas a la base de datos para obtener datos como:
\begin{itemize}
    \item Número total de usuarios registrados.
    \item Número de consultas traducidas.
    \item Tipos de sentencias SQL más comunes (\texttt{SELECT}, \texttt{INSERT}, etc.).
    \item Actividad de los usuarios a lo largo del tiempo.
\end{itemize}
A continuación, se puede observar la implementación del endpoint \texttt{/admin/stats}, accesible solo para administradores, que devuelve estas estadísticas en formato JSON.

    \begin{lstlisting}[language=Python]
        @router.get("/stats")
        def get_admin_stats(
            db: Session = Depends(database.get_db),
            current_user: User = Depends(deps.get_current_user),
        ):
            total_users = db.query(User).count()
            total_translations = db.query(Translation).count()
            failed_translations = (
                db.query(Translation).filter(Translation.error_message is not None).count()
            )
            error_rate = 0
            if total_translations > 0:
                error_rate = (failed_translations / total_translations) * 100

            active_connections = db.query(DbConnection).count()
            recent_activity = (
                db.query(Translation).order_by(Translation.created_at.desc()).limit(5).all()
            )

            return {
                "kpis": {
                    "users": total_users,
                    "translations": total_translations,
                    "error_rate": round(error_rate, 2),
                    "connections": active_connections,
                },
                "recent_activity": recent_activity,
            }
    \end{lstlisting}
